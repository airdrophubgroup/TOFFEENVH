CREATE OR REPLACE FUNCTION join_or_create_match(p_address TEXT) RETURNS JSON AS $$ DECLARE found_match_id UUID; BEGIN SELECT id INTO found_match_id FROM matches WHERE status = 'waiting' AND p1_wallet != p_address LIMIT 1; IF found_match_id IS NOT NULL THEN UPDATE matches SET p2_wallet = p_address, p2_ready = TRUE, status = 'active' WHERE id = found_match_id; RETURN json_build_object('match_id', found_match_id, 'is_p1', false); ELSE INSERT INTO matches (p1_wallet, p1_ready, status) VALUES (p_address, TRUE, 'waiting') RETURNING id INTO found_match_id; RETURN json_build_object('match_id', found_match_id, 'is_p1', true); END IF; END; $$ LANGUAGE plpgsql;
